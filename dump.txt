name: Deploy to Cloudflare Pages

on:
  push:
    branches: [main]
  repository_dispatch:
    types: [data-updated]
  workflow_dispatch:

jobs:
  build-deploy:
    runs-on: ubuntu-latest
    permissions:
      contents: read
      deployments: write

    steps:
      - name: Checkout qimochi
        uses: actions/checkout@v7

      - name: Checkout yukio-data
        uses: actions/checkout@v7
        with:
          repository: bimaakbar-dev/yukio-data
          path: _data

      - name: Inject data
        run: |
          rm -rf src/content/anime
          rm -rf src/data/anime
          rm -rf src/data/actors
          mkdir -p src/content/anime
          mkdir -p src/data
          if [ -d _data/src/content/anime ]; then
            cp -r _data/src/content/anime/. src/content/anime/
          fi
          if [ -d _data/data/anime ]; then
            cp -r _data/data/anime src/data/anime
          fi
          if [ -d _data/data/actors ]; then
            cp -r _data/data/actors src/data/actors
          fi
          echo "=== src/content/anime ==="
          ls -1 src/content/anime/ | wc -l
          echo "=== src/data/anime ==="
          ls -1 src/data/anime/ 2>/dev/null | wc -l

      - name: Setup Node
        uses: actions/setup-node@v7
        with:
          node-version: 24
          cache: npm

      - name: Install dependencies
        run: npm ci

      - name: Build
        run: npm run build
        env:
          PUBLIC_GA_ID: ${{ vars.PUBLIC_GA_ID }}
          PUBLIC_API_BASE: ${{ vars.PUBLIC_API_BASE }}
          PUBLIC_TURNSTILE_SITE_KEY: ${{ vars.PUBLIC_TURNSTILE_SITE_KEY }}
          PUBLIC_GOOGLE_CLIENT_ID: ${{ vars.PUBLIC_GOOGLE_CLIENT_ID }}

      - name: Deploy to Cloudflare Pages
        uses: cloudflare/wrangler-action@v4
        with:
          apiToken: ${{ secrets.CF_API_TOKEN }}
          accountId: ${{ secrets.CF_ACCOUNT_ID }}
          command: pages deploy dist --project-name=qimochi --branch=main

===== FILE: .github/workflows/dump.yml =====
name: Dump repo

on:
  workflow_dispatch:
    inputs:
      max_kb:
        description: Batas ukuran per file (KB)
        default: '200'

permissions:
  contents: write

jobs:
  dump:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v7

      - name: Gabungkan isi repo
        env:
          MAX_KB: ${{ inputs.max_kb }}
        run: |
          OUT=dump.txt
          MAX=$(( ${MAX_KB:-200} * 1024 ))
          SKIP='(^|/)(package-lock\.json|pnpm-lock\.yaml|yarn\.lock|bun\.lockb?)$|^src/content/anime/|^src/data/anime/'

          {
            echo "# Repo: $GITHUB_REPOSITORY"
            echo "# Commit: $GITHUB_SHA"
            echo
            echo "## Struktur file"
            git ls-files | grep -Ev "$SKIP"
            echo
            echo "## Isi file"
          } > "$OUT"

          : > skipped.txt

          git ls-files -z | while IFS= read -r -d '' f; do
            if echo "$f" | grep -Eq "$SKIP"; then continue; fi
            [ -f "$f" ] || continue

            if ! grep -Iq . "$f"; then
              echo "biner atau kosong: $f" >> skipped.txt
              continue
            fi

            size=$(wc -c < "$f")
            if [ "$size" -gt "$MAX" ]; then
              echo "terlalu besar ($size byte): $f" >> skipped.txt
              continue
            fi

            {
              echo
              echo "===== FILE: $f ====="
              cat "$f"
            } >> "$OUT"
          done

          {
            echo
            echo "## File yang dilewati"
            cat skipped.txt
          } >> "$OUT"

          {
            echo "### Dump selesai"
            echo "Ukuran: $(wc -c < "$OUT") byte"
            echo "Dilewati: $(wc -l < skipped.txt) file"
          } >> "$GITHUB_STEP_SUMMARY"

      - name: Simpan hasil ke branch dump
        run: |
          git config user.name "github-actions[bot]"
          git config user.email "41898282+github-actions[bot]@users.noreply.github.com"

          git checkout --orphan dump
          git rm -rf --quiet .
          git add dump.txt
          git commit -m "Dump repo @ ${GITHUB_SHA::7}"
          git push --force origin dump

          echo "File: https://github.com/$GITHUB_REPOSITORY/blob/dump/dump.txt" >> "$GITHUB_STEP_SUMMARY"

<script>
  (function () {
    function setupCoverImage(img: HTMLImageElement): void {
      if (img.dataset.ready) return;
      img.dataset.ready = '1';

      const wrap = img.parentElement;
      const placeholder = wrap?.querySelector<HTMLElement>('.cover-placeholder') ?? null;

      const stopShimmer = () => {
        placeholder?.classList.remove('is-loading');
      };

      const done = () => {
        img.classList.add('is-loaded');
        stopShimmer();
      };

      const fail = () => {
        img.hidden = true;
        stopShimmer();
      };

      if (img.complete) {
        if (img.naturalWidth > 0) done();
        else fail();
        return;
      }

      img.addEventListener('load', done, { once: true });
      img.addEventListener('error', fail, { once: true });
    }

    function init(scope: ParentNode = document): void {
      scope
        .querySelectorAll<HTMLImageElement>('[data-cover-img]')
        .forEach(setupCoverImage);
    }

    init();

    window.addEventListener('qimochi:covers', (e) => {
      const detail = (e as CustomEvent<ParentNode>).detail;
      if (detail) init(detail);
    });
  })();
</script>

  interface AuthUser {
    sub: string;
    name: string;
    email: string;
    picture: string | null;

  (function () {
    const root = document.querySelector<HTMLElement>('[data-rating-widget]');
    if (!root) return;

    const ANIME_ID = root.dataset.anime || '';
    const API = (root.dataset.api || '').replace(/\/$/, '');
    if (!ANIME_ID || !API) return;

    const stars = Array.from(
      root.querySelectorAll<HTMLButtonElement>('[data-star]')
    );
    const scoreBtn = root.querySelector<HTMLButtonElement>('[data-score-btn]')!;
    const averageEl = root.querySelector<HTMLElement>('[data-average]')!;
    const votesEl = root.querySelector<HTMLElement>('[data-votes]')!;
    const userEl = root.querySelector<HTMLElement>('[data-user]')!;
    const clearBtn = root.querySelector<HTMLButtonElement>('[data-clear]')!;
    const distEl = root.querySelector<HTMLElement>('[data-dist]')!;
    const fills: Record<number, HTMLElement> = {};
    const counts: Record<number, HTMLElement> = {};
    for (let n = 1; n <= 5; n++) {
      fills[n] = root.querySelector<HTMLElement>(`[data-fill="${n}"]`)!;
      counts[n] = root.querySelector<HTMLElement>(`[data-count="${n}"]`)!;
    }

    let userScore: number | null = null;
    let currentUser: AuthUser | null = null;
    let submitting = false;
    let distOpen = false;

    function formatAverage(avg: number): string {
      if (!avg) return '—';
      return avg.toFixed(1);
    }

    function paintStars(active: number | null, preview = false): void {
      for (const s of stars) {
        const n = Number(s.dataset.star);
        s.classList.remove('is-active', 'is-preview');
        if (active !== null && n <= active) {
          s.classList.add(preview ? 'is-preview' : 'is-active');
        }
      }
    }

    function updateAria(): void {
      for (const s of stars) {
        const n = Number(s.dataset.star);
        s.setAttribute('aria-checked', String(n === userScore));
      }
    }

    function setInteractive(): void {
      const canRate = !!currentUser;
      for (const s of stars) s.disabled = !canRate || submitting;
      clearBtn.hidden = !(currentUser && userScore !== null);
    }

    function renderUserScore(): void {
      if (userScore !== null) {
        paintStars(userScore);
        userEl.textContent = `kamu: ${userScore}`;
        userEl.hidden = false;
      } else {
        paintStars(null);
        userEl.hidden = true;
      }
      updateAria();
    }

    function renderDistribution(dist: Record<number, number>): void {
      let total = 0;
      for (let n = 1; n <= 5; n++) total += dist[n] || 0;

      if (total === 0) {
        distEl.hidden = true;
        return;
      }

      let maxCount = 0;
      for (let n = 1; n <= 5; n++) {
        if (dist[n] > maxCount) maxCount = dist[n];
      }

      for (let n = 1; n <= 5; n++) {
        const c = dist[n] || 0;
        counts[n].textContent = String(c);
        const pct = maxCount > 0 ? (c / maxCount) * 100 : 0;
        fills[n].style.width = `${pct}%`;
      }

      if (distOpen) distEl.hidden = false;
    }

    function toggleDist(): void {
      if (distEl.hidden && !distOpen) {
        distOpen = true;
        // only show if there is data
        const anyCount = [1, 2, 3, 4, 5].some(
          (n) => Number(counts[n].textContent || '0') > 0
        );
        if (anyCount) distEl.hidden = false;
      } else {
        distOpen = false;
        distEl.hidden = true;
      }
    }

    async function fetchRating(): Promise<void> {
      try {
        const res = await fetch(
          `${API}/api/v1/ratings/${encodeURIComponent(ANIME_ID)}`,
          { credentials: 'include' }
        );
        if (!res.ok) throw new Error(`HTTP ${res.status}`);
        const json = (await res.json()) as {
          data: {
            average: number;
            votes: number;
            userScore: number | null;
            canRate: boolean;
            distribution: Record<number, number>;
          };
        };
        const d = json.data;
        averageEl.textContent = formatAverage(d.average);
        votesEl.textContent = d.votes ? `(${d.votes})` : '';
        userScore = d.userScore;
        renderUserScore();
        setInteractive();
        scoreBtn.disabled = d.votes === 0;
        renderDistribution(d.distribution || {});
      } catch (err) {
        console.error('[rating] fetch failed', err);
        averageEl.textContent = '—';
        votesEl.textContent = '';
        setInteractive();
        scoreBtn.disabled = true;
      }
    }

    async function submitRating(score: number): Promise<void> {
      if (!currentUser || submitting) return;
      submitting = true;
      setInteractive();

      try {
        const res = await fetch(
          `${API}/api/v1/ratings/${encodeURIComponent(ANIME_ID)}`,
          {
            method: 'POST',
            credentials: 'include',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ score }),
          }
        );

        if (res.status === 401) {
          currentUser = null;
          setInteractive();
          return;
        }

        if (!res.ok) throw new Error(`HTTP ${res.status}`);
        const json = (await res.json()) as {
          data: {
            average: number;
            votes: number;
            userScore: number | null;
            distribution: Record<number, number>;
          };
        };
        const d = json.data;
        averageEl.textContent = formatAverage(d.average);
        votesEl.textContent = d.votes ? `(${d.votes})` : '';
        userScore = d.userScore;
        renderUserScore();
        scoreBtn.disabled = d.votes === 0;
        renderDistribution(d.distribution || {});
      } catch (err) {
        console.error('[rating] submit failed', err);
      } finally {
        submitting = false;
        setInteractive();
      }
    }

    async function clearRating(): Promise<void> {
      if (!currentUser || submitting) return;
      submitting = true;

      } finally {

<script>
  interface AuthUser {
    sub: string;
    name: string;
    email: string;
    picture: string | null;
  }

  (function () {
    const root = document.querySelector<HTMLElement>('[data-reaction-bar]');
    if (!root) return;

    const ANIME_ID = root.dataset.anime || '';
    const API = (root.dataset.api || '').replace(/\/$/, '');
    if (!ANIME_ID || !API) return;

    let currentUser: AuthUser | null = null;
    let allowed: string[] = ['👍', '❤️', '🔥', '😂', '😮'];
    let counts: Record<string, number> = {};
    let userReactions: Set<string> = new Set();
    let canReact = false;
    let submitting = false;
    let buttons: Record<string, HTMLButtonElement> = {};

    function buildButtons(): void {
      for (const btn of Object.values(buttons)) btn.remove();
      buttons = {};
      root!.innerHTML = '';

      for (const emoji of allowed) {
        const btn = document.createElement('button');
        btn.type = 'button';
        btn.className = 'reaction-btn';
        btn.dataset.emoji = emoji;
        btn.setAttribute('aria-label', `Reaksi ${emoji}`);
        btn.innerHTML = `
          <span class="reaction-emoji">${emoji}</span>
          <span class="reaction-count" data-count data-empty="true">0</span>
        `;
        btn.addEventListener('click', () => toggle(emoji));
        root!.appendChild(btn);
        buttons[emoji] = btn;
      }
    }

    function applyState(): void {
      for (const emoji of allowed) {
        const btn = buttons[emoji];
        if (!btn) continue;

        const countEl = btn.querySelector<HTMLElement>('[data-count]')!;
        const count = counts[emoji] ?? 0;
        const active = userReactions.has(emoji);

        countEl.textContent = count > 99 ? '99+' : String(count);
        countEl.dataset.empty = count === 0 ? 'true' : 'false';

        btn.classList.toggle('is-active', active);
        btn.disabled = !canReact || submitting;

        btn.setAttribute('aria-label', `Reaksi ${emoji}, ${count} orang`);
        btn.setAttribute('aria-pressed', String(active));
      }
    }

    function popButton(emoji: string): void {
      const btn = buttons[emoji];
      if (!btn) return;
      btn.classList.remove('pop');
      void btn.offsetWidth;
      btn.classList.add('pop');
      window.setTimeout(() => btn.classList.remove('pop'), 400);
    }

    async function fetchReactions(): Promise<void> {
      try {
        const res = await fetch(
          `${API}/api/v1/reactions/${encodeURIComponent(ANIME_ID)}`,
          { credentials: 'include' }
        );
        if (!res.ok) throw new Error(`HTTP ${res.status}`);
        const json = (await res.json()) as {
          data: {
            counts: Record<string, number>;
            userReactions: string[];
            canReact: boolean;
            allowed: string[];
          };
        };
        const d = json.data;
        counts = d.counts || {};
        userReactions = new Set(d.userReactions || []);
        canReact = d.canReact;
        if (Array.isArray(d.allowed) && d.allowed.length) {
          allowed = d.allowed;
        }
        if (Object.keys(buttons).length !== allowed.length) {
          buildButtons();
        }
        applyState();
      } catch (err) {
        console.error('[reaction] fetch failed', err);
      }
    }

    async function toggle(emoji: string): Promise<void> {
      if (!canReact || submitting) return;

      const wasActive = userReactions.has(emoji);
      userReactions = new Set(userReactions);
      if (wasActive) userReactions.delete(emoji);
      else userReactions.add(emoji);

      counts = { ...counts };
      counts[emoji] = Math.max(0, (counts[emoji] ?? 0) + (wasActive ? -1 : 1));

      submitting = true;
      applyState();
      popButton(emoji);

      try {
        const res = await fetch(
          `${API}/api/v1/reactions/${encodeURIComponent(ANIME_ID)}`,
          {
            method: 'POST',
            credentials: 'include',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ emoji }),
          }
        );

        if (res.status === 401) {
          canReact = false;
          await fetchReactions();
          return;
        }

        if (!res.ok) throw new Error(`HTTP ${res.status}`);
        const json = (await res.json()) as {
          data: {
            counts: Record<string, number>;
            userReactions: string[];
          };
        };
        counts = json.data.counts || {};
        userReactions = new Set(json.data.userReactions || []);
      } catch (err) {
        console.error('[reaction] toggle failed', err);
        await fetchReactions();
      } finally {
        submitting = false;
        applyState();
      }
    }

    function readCache(): AuthUser | null {
      try {
        const raw = localStorage.getItem('qimochi:user-cache');
        if (!raw) return null;
        const parsed = JSON.parse(raw) as Partial<AuthUser> | null;
        if (
          parsed &&
          typeof parsed.sub === 'string' &&
          typeof parsed.name === 'string'
        ) {
          return {
            sub: parsed.sub,
            name: parsed.name,
            email: typeof parsed.email === 'string' ? parsed.email : '',
            picture:
              typeof parsed.picture === 'string' ? parsed.picture : null,
          };
        }
      } catch {}
      return null;
    }

    function onAuthChanged(e: Event): void {
      const detail = (e as CustomEvent<{ user: AuthUser | null }>).detail;
      if (!detail) return;
      const same =
        (detail.user?.sub ?? null) === (currentUser?.sub ?? null);
      if (same) return;
      currentUser = detail.user;
      fetchReactions();
    }

    document.addEventListener('qimochi:auth-changed', onAuthChanged);

    (function bootstrap(): void {
      currentUser = readCache();
      buildButtons();
    })();

    fetchReactions();

    (async () => {
      const menu = (window as any).__qimochiUserMenu;
      if (menu) {
        const u = menu.getUser();
        if ((u?.sub ?? null) !== (currentUser?.sub ?? null)) {
          currentUser = u ?? null;
          fetchReactions();
        }
      }
    })();
  })();
</script>

<script>
  interface AuthUser {
    sub: string;
    name: string;
    email: string;
    picture: string | null;
  }

  interface CommentItem {
    id: string;
    parentId: string | null;
    body: string;
    status: string;
    createdAt: number;
    editedAt: number | null;
    likes: number;
    isLiked: boolean;
    isOwn: boolean;
    isAdmin: boolean;
    canEdit: boolean;
    user: { sub: string; name: string; picture: string | null };
  }

  type SortKey = 'oldest' | 'newest' | 'popular';

  (function () {
    const CACHE_KEY = 'qimochi:user-cache';
    const SORT_KEY = 'qimochi:comment-sort';
    const COOLDOWN_SEC = 15;
    const PAGE_SIZE = 10;

    const root = document.querySelector<HTMLElement>('[data-comments]');
    if (!root) return;

    const SLUG = root.dataset.slug || '';
    const API = (root.dataset.api || '').replace(/\/$/, '');
    const SITE_KEY = root.dataset.sitekey || '';

    const countEl = root.querySelector<HTMLElement>('[data-count]')!;
    const skeletonBox = root.querySelector<HTMLElement>('[data-auth-skeleton]')!;
    const loginBox = root.querySelector<HTMLElement>('[data-login-box]')!;
    const gsiSlot = root.querySelector<HTMLElement>('[data-gsi-slot]')!;
    const userBox = root.querySelector<HTMLElement>('[data-user-box]')!;
    const userAvatar = root.querySelector<HTMLImageElement>('[data-user-avatar]')!;
    const userName = root.querySelector<HTMLElement>('[data-user-name]')!;
    const logoutBtn = root.querySelector<HTMLButtonElement>('[data-logout]')!;
    const form = root.querySelector<HTMLFormElement>('[data-form]')!;
    const replyContext = root.querySelector<HTMLElement>('[data-reply-context]')!;
    const replyLabel = root.querySelector<HTMLElement>('[data-reply-label]')!;
    const replyCancel = root.querySelector<HTMLButtonElement>('[data-reply-cancel]')!;
    const input = root.querySelector<HTMLTextAreaElement>('[data-input]')!;
    const charCount = root.querySelector<HTMLElement>('[data-charcount]')!;
    const submitBtn = root.querySelector<HTMLButtonElement>('[data-submit]')!;
    const formStatus = root.querySelector<HTMLElement>('[data-form-status]')!;
    const toolbar = root.querySelector<HTMLElement>('[data-toolbar]')!;
    const sortSelect = root.querySelector<HTMLSelectElement>('[data-sort]')!;
    const statusEl = root.querySelector<HTMLElement>('[data-status]')!;
    const listEl = root.querySelector<HTMLOListElement>('[data-list]')!;
    const loadMoreWrap = root.querySelector<HTMLElement>('[data-load-more-wrap]')!;
    const loadMoreBtn = root.querySelector<HTMLButtonElement>('[data-load-more]')!;
    const tmpl = root.querySelector<HTMLTemplateElement>('[data-item-tmpl]')!;

    if (!SLUG || !API) {
      statusEl.textContent = 'Konfigurasi komentar tidak lengkap.';
      return;
    }

    let currentUser: AuthUser | null = null;
    let allComments: CommentItem[] = [];
    let rootComments: CommentItem[] = [];
    let childrenMap: Record<string, CommentItem[]> = {};
    let visibleRoots = PAGE_SIZE;
    let sortKey: SortKey = 'oldest';
    let turnstileWidgetId: string | null = null;
    let turnstileToken = '';
    let submitting = false;
    let gsiRendered = false;
    let replyTarget: { id: string; name: string } | null = null;
    let cooldownUntil = 0;
    let cooldownTimer: number | null = null;
    let highlightId: string | null = null;

    try {
      const saved = localStorage.getItem(SORT_KEY) as SortKey | null;
      if (saved === 'oldest' || saved === 'newest' || saved === 'popular') {
        sortKey = saved;
      }
    } catch {}

    sortSelect.value = sortKey;

    function readCache(): AuthUser | null {
      try {
        const raw = localStorage.getItem(CACHE_KEY);
        if (!raw) return null;
        const parsed = JSON.parse(raw) as Partial<AuthUser> | null;
        if (
          parsed &&
          typeof parsed.sub === 'string' &&
          typeof parsed.name === 'string'
        ) {
          return {
            sub: parsed.sub,
            name: parsed.name,
            email: typeof parsed.email === 'string' ? parsed.email : '',
            picture:
              typeof parsed.picture === 'string' ? parsed.picture : null,
          };
        }
      } catch {}
      return null;
    }

    function loadScript(src: string): Promise<void> {
      return new Promise((resolve, reject) => {
        const existing = document.querySelector<HTMLScriptElement>(
          `script[src="${src}"]`
        );
        if (existing) {
          if (existing.dataset.loaded === '1') return resolve();
          existing.addEventListener('load', () => resolve(), { once: true });
          existing.addEventListener('error', () => reject(), { once: true });
          return;
        }
        const s = document.createElement('script');
        s.src = src;
        s.async = true;
        s.defer = true;
        s.addEventListener(
          'load',
          () => {
            s.dataset.loaded = '1';
            resolve();
          },
          { once: true }
        );
        s.addEventListener('error', () => reject(), { once: true });
        document.head.appendChild(s);
      });
    }

    function escapeHtml(s: string): string {
      return s
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;')
        .replace(/'/g, '&#39;');
    }

    function renderMarkdown(text: string): string {
      let s = escapeHtml(text);

      s = s.replace(
        /\|\|([^|\n]+?)\|\|/g,
        '<span class="spoiler" data-spoiler>$1</span>'
      );

      s = s.replace(/`([^`\n]+)`/g, '<code>$1</code>');
      s = s.replace(/\*\*([^*\n]+)\*\*/g, '<strong>$1</strong>');
      s = s.replace(/(^|[^*])\*([^*\n]+)\*(?!\*)/g, '$1<em>$2</em>');
      s = s.replace(/~~([^~\n]+)~~/g, '<del>$1</del>');

      s = s.replace(
        /(^|\s)@([A-Za-z0-9_]{2,30})/g,
        '$1<span class="mention">@$2</span>'
      );

      s = s.replace(
        /\[([^\]]+)\]\((https?:\/\/[^\s)]+)\)/g,
        '<a href="$2" target="_blank" rel="nofollow ugc noopener noreferrer">$1</a>'
      );

      s = s.replace(
        /(^|[^"=])(https?:\/\/[^\s<]+)/g,
        (_, prefix, url) =>
          `${prefix}<a href="${url}" target="_blank" rel="nofollow ugc noopener noreferrer">${url}</a>`
      );

      s = s.replace(/\n/g, '<br>');

      return s;
    }

    function formatTime(ts: number): string {
      const d = new Date(ts * 1000);
      const now = Date.now();
      const diff = Math.floor((now - d.getTime()) / 1000);
      if (diff < 60) return 'baru saja';
      if (diff < 3600) return `${Math.floor(diff / 60)} menit lalu`;
      if (diff < 86400) return `${Math.floor(diff / 3600)} jam lalu`;
      if (diff < 604800) return `${Math.floor(diff / 86400)} hari lalu`;
      return new Intl.DateTimeFormat('id-ID', {
        day: 'numeric',
        month: 'short',
        year: 'numeric',
      }).format(d);
    }

    function statusBadgeText(status: string): string {
      if (status === 'pending') return 'Menunggu moderasi';
      if (status === 'rejected') return 'Ditolak';
      return '';
    }

    function sortRoots(items: CommentItem[]): CommentItem[] {
      const arr = [...items];
      if (sortKey === 'newest') {
        arr.sort((a, b) => b.createdAt - a.createdAt);
      } else if (sortKey === 'popular') {
        arr.sort((a, b) => {
          if (b.likes !== a.likes) return b.likes - a.likes;
          return b.createdAt - a.createdAt;
        });
      } else {
        arr.sort((a, b) => a.createdAt - b.createdAt);
      }
      return arr;
    }

    function makeItem(item: CommentItem, isReply: boolean): HTMLLIElement {
      const node = tmpl.content.firstElementChild!.cloneNode(
        true
      ) as HTMLLIElement;
      node.dataset.id = item.id;
      node.dataset.parentId = item.parentId ?? '';

      const avatar = node.querySelector<HTMLImageElement>('[data-avatar]')!;
      if (item.user.picture) {
        avatar.src = item.user.picture;
        avatar.alt = item.user.name;
      } else {
        avatar.removeAttribute('src');
        avatar.style.background = 'var(--bg-surface)';
      }

      node.querySelector<HTMLElement>('[data-name]')!.textContent =
        item.user.name;

      const roleEl = node.querySelector<HTMLElement>('[data-role]')!;
      if (item.isAdmin) {
        roleEl.hidden = false;
      }

      const timeEl = node.querySelector<HTMLTimeElement>('[data-time]')!;
      timeEl.dateTime = new Date(item.createdAt * 1000).toISOString();
      timeEl.textContent = formatTime(item.createdAt);

      const editedEl = node.querySelector<HTMLElement>('[data-edited]')!;
      if (item.editedAt) editedEl.hidden = false;

      const bodyEl = node.querySelector<HTMLElement>('[data-body]')!;
      bodyEl.innerHTML = renderMarkdown(item.body);

      bodyEl.querySelectorAll<HTMLElement>('[data-spoiler]').forEach((el) => {
        el.addEventListener('click', () => {
          el.classList.toggle('revealed');
        });
      });

      const badge = node.querySelector<HTMLElement>('[data-badge]')!;
      if (item.isOwn && item.status !== 'approved') {
        const text = statusBadgeText(item.status);
        if (text) {
          badge.textContent = text;
          badge.classList.add(
            item.status === 'rejected' ? 'is-rejected' : 'is-pending'
          );
          badge.hidden = false;
        }
      }

      const likeBtn = node.querySelector<HTMLButtonElement>(
        '[data-action="like"]'
      )!;
      likeBtn.dataset.id = item.id;

      const likeIcon = likeBtn.querySelector<HTMLElement>('[data-like-icon]')!;
      const likeCount = likeBtn.querySelector<HTMLElement>('[data-like-count]')!;

      likeIcon.textContent = item.isLiked ? '♥' : '♡';
      if (item.likes > 0) {
        likeCount.textContent = String(item.likes);
        likeCount.hidden = false;
      }
      if (item.isLiked) likeBtn.classList.add('is-liked');
      if (!currentUser) likeBtn.hidden = true;

      const replyBtn = node.querySelector<HTMLButtonElement>(
        '[data-action="reply"]'
      )!;
      if (isReply || !currentUser) {
        replyBtn.hidden = true;
      } else {
        replyBtn.dataset.id = item.id;
        replyBtn.dataset.name = item.user.name;
      }

      const editBtn = node.querySelector<HTMLButtonElement>(
        '[data-action="edit"]'
      )!;
      if (item.canEdit) {
        editBtn.hidden = false;
        editBtn.dataset.id = item.id;
      }

      const deleteBtn = node.querySelector<HTMLButtonElement>(
        '[data-action="delete"]'
      )!;
      if (item.isOwn) {
        deleteBtn.hidden = false;
        deleteBtn.dataset.id = item.id;
        if (item.parentId === null) deleteBtn.dataset.root = '1';
      }

      const reportBtn = node.querySelector<HTMLButtonElement>(
        '[data-action="report"]'
      )!;
      if (currentUser && !item.isOwn) {
        reportBtn.hidden = false;
        reportBtn.dataset.id = item.id;
      }

      const editBox = node.querySelector<HTMLElement>('[data-edit-box]')!;
      const editInput = node.querySelector<HTMLTextAreaElement>(
        '[data-edit-input]'
      )!;
      editInput.value = item.body;
      editBox.dataset.id = item.id;

      return node;
    }

    function render(): void {
      const visible = rootComments.slice(0, visibleRoots);

      listEl.innerHTML = '';
      for (const r of visible) {
        const node = makeItem(r, false);
        const replies = childrenMap[r.id] ?? [];
        if (replies.length > 0) {
          const repliesEl = node.querySelector<HTMLOListElement>(
            '[data-replies]'
          )!;
          for (const reply of replies) {
            repliesEl.appendChild(makeItem(reply, true));
          }
          repliesEl.hidden = false;
        }
        listEl.appendChild(node);
      }

      countEl.textContent = String(allComments.length);
      listEl.hidden = visible.length === 0;

      const remaining = rootComments.length - visible.length;
      if (remaining > 0) {
        loadMoreWrap.hidden = false;
        loadMoreBtn.textContent = `Muat lebih banyak (${remaining})`;
      } else {
        loadMoreWrap.hidden = true;
      }

      toolbar.hidden = rootComments.length === 0;

      if (highlightId) {
        const target = listEl.querySelector<HTMLElement>(
          `[data-id="${highlightId}"]`
        );
        if (target) {
          target.classList.add('just-posted');
          target.scrollIntoView({ behavior: 'smooth', block: 'center' });
          window.setTimeout(() => {
            target.classList.remove('just-posted');
          }, 2500);
          highlightId = null;
        }
      }
    }

    async function fetchComments(): Promise<void> {
      statusEl.hidden = false;
      statusEl.textContent = 'Memuat komentar…';
      listEl.hidden = true;
      try {
        const res = await fetch(
          `${API}/api/v1/comments?slug=${encodeURIComponent(SLUG)}`,
          { credentials: 'include' }
        );
        if (!res.ok) throw new Error(`HTTP ${res.status}`);
        const json = (await res.json()) as { data: CommentItem[] };
        allComments = Array.isArray(json.data) ? json.data : [];

        rootComments = allComments.filter((c) => c.parentId === null);
        childrenMap = {};
        for (const c of allComments) {
          if (c.parentId) {
            (childrenMap[c.parentId] ??= []).push(c);
          }
        }

        for (const key of Object.keys(childrenMap)) {
          childrenMap[key]!.sort((a, b) => a.createdAt - b.createdAt);
        }

        rootComments = sortRoots(rootComments);
        visibleRoots = Math.max(PAGE_SIZE, visibleRoots);

        if (allComments.length === 0) {
          statusEl.textContent = 'Belum ada komentar. Jadilah yang pertama.';
          listEl.hidden = true;
          loadMoreWrap.hidden = true;
          toolbar.hidden = true;
          countEl.textContent = '0';
          return;
        }

        render();
        statusEl.hidden = true;
      } catch (err) {
        console.error('[comments] fetch failed', err);
        statusEl.textContent = 'Gagal memuat komentar. Coba refresh.';
      }
    }

    function renderGsiButton(): void {
      if (gsiRendered) return;
      if (currentUser) return;
      const g = (window as any).google;
      if (!g?.accounts?.id) return;
      try {
        g.accounts.id.renderButton(gsiSlot, {
          theme: 'outline',
          size: 'large',
          text: 'signin_with',
          shape: 'pill',
        });
        gsiRendered = true;
      } catch (err) {
        console.error('[comments] render gsi failed', err);
      }
    }

    function applyUserState(): void {
      skeletonBox.hidden = true;

      if (currentUser) {
        loginBox.hidden = true;
        userBox.hidden = false;
        form.hidden = false;
        userName.textContent = currentUser.name;
        if (currentUser.picture) {
          userAvatar.src = currentUser.picture;
          userAvatar.alt = currentUser.name;
        } else {
          userAvatar.removeAttribute('src');
        }
        mountTurnstile();
      } else {
        loginBox.hidden = false;
        userBox.hidden = true;
        form.hidden = true;
        gsiRendered = false;
        gsiSlot.innerHTML = '';
        renderGsiButton();
      }
    }

    function mountTurnstile(): void {
      if (!SITE_KEY) return;
      if (turnstileWidgetId) return;
      const container = root!.querySelector<HTMLElement>('[data-turnstile]');
      if (!container) return;
      const ts = (window as any).turnstile;
      if (!ts) return;
      try {
        turnstileWidgetId = ts.render(container, {
          sitekey: SITE_KEY,
          callback: (token: string) => {
            turnstileToken = token;
            updateSubmitState();
          },
          'expired-callback': () => {
            turnstileToken = '';
            updateSubmitState();
          },
          'error-callback': () => {
            turnstileToken = '';
            updateSubmitState();
          },
        });
      } catch (err) {
        console.error('[turnstile] render failed', err);
      }
    }

    function isCooldown(): boolean {
      return Date.now() < cooldownUntil;
    }

    function updateSubmitState(): void {
      if (isCooldown()) return;
      const text = input.value.trim();
      const ok =
        !!currentUser && text.length >= 2 && !!turnstileToken && !submitting;
      submitBtn.disabled = !ok;
      submitBtn.textContent = replyTarget ? 'Balas' : 'Kirim';
    }

    function tickCooldown(): void {
      const remain = cooldownUntil - Date.now();
      if (remain <= 0) {
        if (cooldownTimer !== null) {
          window.clearInterval(cooldownTimer);
          cooldownTimer = null;
        }
        cooldownUntil = 0;
        updateSubmitState();
        return;
      }
      submitBtn.disabled = true;
      submitBtn.textContent = `Tunggu ${Math.ceil(remain / 1000)}s`;
    }

    function startCooldown(): void {
      cooldownUntil = Date.now() + COOLDOWN_SEC * 1000;
      if (cooldownTimer !== null) window.clearInterval(cooldownTimer);
      cooldownTimer = window.setInterval(tickCooldown, 500);
      tickCooldown();
    }

    function setFormStatus(
      text: string,
      kind: 'error' | 'success' | null
    ): void {
      formStatus.textContent = text;
      formStatus.classList.remove('is-error', 'is-success');
      if (kind === 'error') formStatus.classList.add('is-error');
      if (kind === 'success') formStatus.classList.add('is-success');
      formStatus.hidden = !text;
    }

    function setReplyTarget(target: { id: string; name: string } | null): void {
      replyTarget = target;
      if (target) {
        replyLabel.textContent = `Membalas ${target.name}`;
        replyContext.hidden = false;
        form.scrollIntoView({ behavior: 'smooth', block: 'center' });
        window.setTimeout(() => input.focus(), 300);
      } else {
        replyContext.hidden = true;
        replyLabel.textContent = '';
      }
      updateSubmitState();
    }

    function setCurrentUser(user: AuthUser | null): void {
      const sameUser = (user?.sub ?? null) === (currentUser?.sub ?? null);
      if (sameUser) return;
      currentUser = user;
      applyUserState();
      updateSubmitState();
      fetchComments();
    }

    function onAuthChanged(e: Event): void {
      const detail = (e as CustomEvent<{ user: AuthUser | null }>).detail;
      if (!detail) return;
      setCurrentUser(detail.user);
    }

    document.addEventListener('qimochi:auth-changed', onAuthChanged);
    document.addEventListener('qimochi:gsi-ready', () => renderGsiButton());

    if ((window as any).google?.accounts?.id) {
      renderGsiButton();
    }

    logoutBtn.addEventListener('click', async () => {
      const api = (window as any).__qimochiUserMenu;
      if (api) api.close();
      try {
        await fetch(`${API}/api/v1/auth/logout`, {
          method: 'POST',
          credentials: 'include',
        });
      } catch {}
      setCurrentUser(null);
      turnstileToken = '';
      setFormStatus('', null);
      setReplyTarget(null);
      document.dispatchEvent(
        new CustomEvent('qimochi:auth-changed', { detail: { user: null } })
      );
    });

    replyCancel.addEventListener('click', () => setReplyTarget(null));

    sortSelect.addEventListener('change', () => {
      const v = sortSelect.value as SortKey;
      if (v !== 'oldest' && v !== 'newest' && v !== 'popular') return;
      sortKey = v;
      try {
        localStorage.setItem(SORT_KEY, v);

const currentPath = Astro.url.pathname;

function isActive(href: string): boolean {
  if (href === '/') return currentPath === '/';
  const matches = currentPath === href || currentPath.startsWith(href);
  if (!matches) return false;
  const hasMoreSpecific = NAV_LINKS.some(({ href: otherHref }) => {
    if (otherHref === href) return false;
    if (otherHref === '/') return false;
    if (otherHref.length <= href.length) return false;
    return currentPath === otherHref || currentPath.startsWith(otherHref);
  });
  return !hasMoreSpecific;
}

  (function () {
    const KEY = 'qimochi:continue';
    const MAX_SHOW = 6;

    const section = document.querySelector<HTMLElement>('[data-history-section]');
    const list = document.querySelector<HTMLElement>('[data-history-list]');
    const tmpl = document.getElementById('history-item-tmpl') as HTMLTemplateElement | null;

    if (!section || !list || !tmpl) return;

    type Item = {
      slug: string;
      title: string;
      cover: string;
      episode: number;
      total: number;
      ts: number;
    };

    let items: Item[] = [];
    try {
      const raw = localStorage.getItem(KEY);
      if (raw) {
        const parsed = JSON.parse(raw);
        if (Array.isArray(parsed)) {
          items = parsed.filter(
            (x): x is Item =>
              x &&
              typeof x.slug === 'string' &&
              typeof x.title === 'string' &&
              typeof x.episode === 'number'
          );
        }
      }
    } catch {}

    if (items.length === 0) return;

    items.sort((a, b) => (b.ts ?? 0) - (a.ts ?? 0));
    const visible = items.slice(0, MAX_SHOW);

    const frag = document.createDocumentFragment();

    for (const item of visible) {
      const node = tmpl.content.firstElementChild!.cloneNode(true) as HTMLElement;
      const link = node.querySelector('[data-history-link]') as HTMLAnchorElement;
      link.setAttribute('href', `/anime/watch/${item.slug}/episodes/${item.episode}/`);

      (node.querySelector('[data-history-ep]') as HTMLElement).textContent =
        String(item.episode).padStart(2, '0');

      (node.querySelector('[data-history-title]') as HTMLElement).textContent = item.title;

      const meta = node.querySelector('[data-history-meta]') as HTMLElement;
      if (item.total > 0) {
        meta.textContent = `Episode ${item.episode} dari ${item.total}`;
      } else {
        meta.textContent = `Episode ${item.episode}`;
      }

      if (item.total > 0 && item.episode > 0) {
        const progress = node.querySelector('[data-history-progress]') as HTMLElement;
        const fill = node.querySelector('[data-history-progress-fill]') as HTMLElement;
        const pct = Math.min(100, Math.round((item.episode / item.total) * 100));
        fill.style.width = `${pct}%`;
        progress.hidden = false;

        if (item.episode >= item.total) {
          fill.classList.add('is-complete');
        }
      }

      frag.appendChild(node);
    }

    list.appendChild(frag);
    section.hidden = false;
  })();
</script>

<script>
  interface AuthUser {
    sub: string;
    name: string;
    email: string;
    picture: string | null;
  }

  (function () {
    const CACHE_KEY = 'qimochi:user-cache';

    const root = document.querySelector<HTMLElement>('[data-user-menu]');
    if (!root) return;

    const CLIENT_ID = root.dataset.clientid || '';
    const API = (root.dataset.api || '').replace(/\/$/, '');

    const trigger = root.querySelector<HTMLButtonElement>('[data-trigger]')!;
    const guestIcon = root.querySelector<HTMLElement>('[data-guest]')!;
    const triggerAvatar = root.querySelector<HTMLImageElement>('[data-avatar]')!;
    const panel = root.querySelector<HTMLElement>('[data-panel]')!;
    const gsiSlot = root.querySelector<HTMLElement>('[data-gsi]')!;
    const profileBox = root.querySelector<HTMLElement>('[data-profile]')!;
    const infoAvatar = root.querySelector<HTMLImageElement>('[data-info-avatar]')!;
    const infoName = root.querySelector<HTMLElement>('[data-info-name]')!;
    const infoEmail = root.querySelector<HTMLElement>('[data-info-email]')!;
    const logoutBtn = root.querySelector<HTMLButtonElement>('[data-logout]')!;

    let currentUser: AuthUser | null = null;
    let gsiRendered = false;
    let panelOpen = false;

    function readCache(): AuthUser | null {
      try {
        const raw = localStorage.getItem(CACHE_KEY);
        if (!raw) return null;
        const parsed = JSON.parse(raw) as Partial<AuthUser> | null;
        if (parsed && typeof parsed.sub === 'string' && typeof parsed.name === 'string') {
          return {
            sub: parsed.sub,
            name: parsed.name,
            email: typeof parsed.email === 'string' ? parsed.email : '',
            picture: typeof parsed.picture === 'string' ? parsed.picture : null,
          };
        }
      } catch {}
      return null;
    }

    function writeCache(user: AuthUser | null): void {
      try {
        if (user) localStorage.setItem(CACHE_KEY, JSON.stringify(user));
        else localStorage.removeItem(CACHE_KEY);
      } catch {}
    }

    function loadScript(src: string): Promise<void> {
      return new Promise((resolve, reject) => {
        const existing = document.querySelector<HTMLScriptElement>(`script[src="${src}"]`);
        if (existing) {
          if (existing.dataset.loaded === '1') return resolve();
          existing.addEventListener('load', () => resolve(), { once: true });
          existing.addEventListener('error', () => reject(), { once: true });
          return;
        }
        const s = document.createElement('script');
        s.src = src;
        s.async = true;
        s.defer = true;
        s.addEventListener('load', () => { s.dataset.loaded = '1'; resolve(); }, { once: true });
        s.addEventListener('error', () => reject(), { once: true });
        document.head.appendChild(s);
      });
    }

    function dispatchAuthChanged(): void {
      document.dispatchEvent(
        new CustomEvent('qimochi:auth-changed', { detail: { user: currentUser } })
      );
    }

    function applyUser(): void {
      if (currentUser) {
        guestIcon.hidden = true;
        triggerAvatar.hidden = false;
        if (currentUser.picture) {
          triggerAvatar.src = currentUser.picture;
          triggerAvatar.alt = currentUser.name;
        } else {
          triggerAvatar.removeAttribute('src');
        }
        gsiSlot.innerHTML = '';
        gsiSlot.hidden = true;
        profileBox.hidden = false;
        if (currentUser.picture) {
          infoAvatar.src = currentUser.picture;
        } else {
          infoAvatar.removeAttribute('src');
        }
        infoAvatar.alt = currentUser.name;
        infoName.textContent = currentUser.name;
        infoEmail.textContent = currentUser.email;
      } else {
        guestIcon.hidden = false;
        triggerAvatar.hidden = true;
        triggerAvatar.removeAttribute('src');
        profileBox.hidden = true;
        gsiSlot.hidden = false;
        ensureGsi();
      }
    }

    function ensureGsi(): void {
      if (gsiRendered) return;
      if (!CLIENT_ID) return;
      const g = (window as any).google;
      if (!g?.accounts?.id) return;
      if (!panelOpen) return;

      try {
        g.accounts.id.renderButton(gsiSlot, {
          theme: 'outline',
          size: 'large',
          text: 'signin_with',
          shape: 'rect',
          logo_alignment: 'left',
          width: 240,
        });
        gsiRendered = true;
      } catch (err) {
        console.error('[usermenu] render gsi failed', err);
      }
    }

    function openPanel(): void {
      panelOpen = true;
      panel.hidden = false;
      trigger.setAttribute('aria-expanded', 'true');
      if (!currentUser) ensureGsi();
    }

    function closePanel(): void {
      panelOpen = false;
      panel.hidden = true;
      trigger.setAttribute('aria-expanded', 'false');
    }

    trigger.addEventListener('click', (e) => {
      e.stopPropagation();
      if (panelOpen) closePanel();
      else openPanel();
    });

    document.addEventListener('click', (e) => {
      if (!panelOpen) return;
      if (root.contains(e.target as Node)) return;
      closePanel();
    });

    document.addEventListener('keydown', (e) => {
      if (e.key === 'Escape' && panelOpen) closePanel();
    });

    async function handleCredential(response: { credential: string }): Promise<void> {
      try {
        const res = await fetch(`${API}/api/v1/auth/google`, {
          method: 'POST',
          credentials: 'include',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ credential: response.credential }),
        });
        if (!res.ok) throw new Error(`HTTP ${res.status}`);
        const json = (await res.json()) as { ok: boolean; user: AuthUser };
        if (json.ok && json.user) {
          currentUser = json.user;
          writeCache(json.user);
          applyUser();
          closePanel();
          dispatchAuthChanged();
        }
      } catch (err) {
        console.error('[usermenu] login failed', err);
      }
    }

    logoutBtn.addEventListener('click', async () => {
      try {
        await fetch(`${API}/api/v1/auth/logout`, {
          method: 'POST',
          credentials: 'include',
        });
      } catch {}
      currentUser = null;
      gsiRendered = false;
      writeCache(null);
      applyUser();
      closePanel();
      dispatchAuthChanged();
    });

    async function fetchMe(): Promise<void> {
      try {
        const res = await fetch(`${API}/api/v1/auth/me`, { credentials: 'include' });
        if (!res.ok) return;
        const json = (await res.json()) as { ok: boolean; user: AuthUser | null };
        const user = json.ok && json.user ? json.user : null;
        const same = (user?.sub ?? null) === (currentUser?.sub ?? null);
        if (same) return;
        currentUser = user;
        writeCache(user);
        applyUser();
        dispatchAuthChanged();
      } catch {
        // biarkan state dari cache
      }
    }

    function onExternalChange(e: Event): void {
      const detail = (e as CustomEvent<{ user: AuthUser | null }>).detail;
      if (!detail) return;
      const sameUser = (detail.user?.sub ?? null) === (currentUser?.sub ?? null);
      if (sameUser) return;
      currentUser = detail.user;
      writeCache(detail.user);
      applyUser();
    }

    document.addEventListener('qimochi:auth-changed', onExternalChange);

    (function bootstrap(): void {
      currentUser = readCache();
      applyUser();
      if (currentUser) dispatchAuthChanged();
    })();

    (async () => {
      if (CLIENT_ID) {
        try {
          await loadScript('https://accounts.google.com/gsi/client');
          const g = (window as any).google;
          if (g?.accounts?.id) {
            g.accounts.id.initialize({
              client_id: CLIENT_ID,
              callback: handleCredential,
              auto_select: false,
              cancel_on_tap_outside: true,
            });
            document.dispatchEvent(new CustomEvent('qimochi:gsi-ready'));
          }
        } catch (err) {
          console.error('[usermenu] gsi load failed', err);
        }
      }

      await fetchMe();
    })();

    (window as any).__qimochiUserMenu = {
      getUser: () => currentUser,
      open: () => openPanel(),
      close: () => closePanel(),
    };
  })();
</script>

Selamat datang di **Qimochi** — tempat untuk streaming dan download anime batch
dengan pengalaman yang ringan, cepat, dan tanpa ribet.

## Apa Itu Qimochi?

Qimochi adalah koleksi anime yang bisa kamu tonton langsung atau download
dalam berbagai kualitas. Kami fokus pada tiga hal utama:

### 1. Kecepatan

Situs dibangun dengan teknologi statis modern. Setiap halaman dimuat dalam
waktu kurang dari 1 detik, bahkan di koneksi lambat.

### 2. Kemudahan

Navigasi simpel — tidak ada popup, tidak ada iklan mengganggu, tidak ada
redirect aneh. Kamu datang untuk nonton, kami pastikan itu yang kamu dapat.

### 3. Kelengkapan

Koleksi anime terus bertambah. Dari judul klasik sampai musim terbaru,
semuanya ada di satu tempat.

## Fitur Utama

- **Streaming langsung** — tonton tanpa download
- **Download batch** — unduh semua episode sekaligus
- **Multi-kualitas** — 360p, 480p, 720p, 1080p
- **Multiple server** — ganti server kalau yang satu lag
- **Pencarian canggih** — cari dengan typo tetap ketemu
- **Archive lengkap** — browse berdasarkan abjad
- **Genre filter** — temukan sesuai selera

## Cara Menggunakan

### 1. Browse Anime

Buka halaman [Archive](/anime/) untuk melihat semua anime. Atau jelajahi
berdasarkan [genre](/genre/) kalau kamu punya preferensi tertentu.

### 2. Pilih Episode

Setiap anime punya halaman detail dengan daftar episode lengkap. Klik
episode yang mau ditonton.

### 3. Pilih Server & Kualitas

Di halaman tonton, kamu bisa memilih:

- **Kualitas** — dari 360p (hemat kuota) sampai 1080p (kualitas terbaik)
- **Server** — ganti-ganti kalau yang satu bermasalah

Pilihanmu akan **diingat otomatis** untuk kunjungan berikutnya.

### 4. Nonton atau Download

- **Nonton langsung** — klik server, video langsung main
- **Download** — pilih kualitas, klik server download

## Yang Perlu Diketahui

Kami **tidak menyimpan file video** di server kami. Semua konten di-embed
dari layanan pihak ketiga yang tersedia bebas di internet. Kami hanya
menyediakan tautan untuk keperluan referensi.

Untuk informasi lebih lanjut, lihat halaman [DMCA](/dmca/) dan
[Disclaimer](/disclaimer/).

## Bergabung dengan Komunitas

Punya pertanyaan, saran, atau menemukan bug? Kami selalu terbuka untuk
masukan. Sampaikan lewat:

- **RSS Feed** — subscribe di [RSS](/rss.xml) untuk update artikel baru
- **Komentar** — (segera hadir)

## Apa Selanjutnya?

Kami terus mengembangkan Qimochi. Beberapa fitur yang sedang dalam
perencanaan:

- Sistem komentar untuk diskusi
- Watchlist pribadi
- Notifikasi episode baru
- Rekomendasi personal

Pantau terus blog ini untuk update terbaru. Selamat menjelajah! 🎉

import { getCollection, getEntry, type CollectionEntry } from 'astro:content';

export interface EpisodeServer {
  name: string;
  url: string;
}

export interface EpisodeStream {
  quality: string;
  servers: EpisodeServer[];
}

export interface EpisodeDownload {
  quality: string;
  size: string;
  servers: EpisodeServer[];
}

export interface EpisodeData {
  number: number;
  title?: string;
  streams: EpisodeStream[];
  downloads?: EpisodeDownload[];
}

export interface Franchise {
  relation: string;
  slug: string;
  title?: string;
}

export type AnimeEntry = CollectionEntry<'anime'>;

type QimochiStatus = 'Ongoing' | 'Completed' | 'Hiatus' | 'Upcoming';
type QimochiType = 'TV' | 'Movie' | 'OVA' | 'ONA' | 'Special';

export type HydratedAnime = Omit<AnimeEntry, 'data'> & {
  data: {
    title: string;
    cover: string;
    status: QimochiStatus;
    type: QimochiType;
    genre: string[];
    studio: string;
    releaseDate: Date;
    addedAt: Date;
    updatedAt?: Date;
    rating: number;
    episodes: EpisodeData[];
    franchises: Franchise[];

    titleEnglish?: string;
    titleNative?: string;
    year?: number;
    malId?: number;
    banner?: string;
    trailer?: string;
    duration?: number;
  };
};

interface ChunkRef {
  slug: string;
  start: number;
  end: number;
  path: string;
}

const HIDDEN_RELATIONS = new Set([
  'character',
  'adaptation',
  'contains',
  'other',
]);

const episodeModules = import.meta.glob<{ default: EpisodeData[] }>(
  '../data/anime/*/episodes/streams/*.json',
  { eager: true }
);

const franchiseModules = import.meta.glob<{ default: Franchise[] }>(
  '../data/anime/*/franchises.json',
  { eager: true }
);

const chunksBySlug: Record<string, ChunkRef[]> = {};

for (const [path, mod] of Object.entries(episodeModules)) {
  const match = path.match(
  /\/data\/anime\/([^/]+)\/episodes\/streams\/(\d+)-(\d+)\.json$/
  );
  if (!match) continue;

  const slug = match[1];
  const start = parseInt(match[2] ?? '0', 10);
  const end = parseInt(match[3] ?? '0', 10);
  if (!slug || isNaN(start) || isNaN(end)) continue;
  if (!Array.isArray(mod.default)) continue;

  if (!chunksBySlug[slug]) chunksBySlug[slug] = [];
  chunksBySlug[slug]!.push({ slug, start, end, path });
}

const episodesBySlug: Record<string, EpisodeData[]> = {};

for (const [slug, chunks] of Object.entries(chunksBySlug)) {
  chunks.sort((a, b) => a.start - b.start);
  const merged: EpisodeData[] = [];
  for (const chunk of chunks) {
    const mod = episodeModules[chunk.path];
    if (!mod || !Array.isArray(mod.default)) continue;
    merged.push(...mod.default);
  }
  merged.sort((a, b) => a.number - b.number);
  episodesBySlug[slug] = merged;
}

const franchisesBySlug: Record<string, Franchise[]> = {};

for (const [path, mod] of Object.entries(franchiseModules)) {
  const match = path.match(/\/data\/anime\/([^/]+)\/franchises\.json$/);
  if (!match) continue;
  const slug = match[1];
  if (!slug) continue;

  const raw = Array.isArray(mod.default) ? mod.default : [];
  franchisesBySlug[slug] = raw.filter(
    (f) => f && typeof f.slug === 'string' && !HIDDEN_RELATIONS.has(f.relation)
  );
}

function mapStatus(s: string): QimochiStatus {
  switch (s) {
    case 'airing':
      return 'Ongoing';
    case 'finished':
      return 'Completed';
    case 'upcoming':
      return 'Upcoming';
    case 'hiatus':
      return 'Hiatus';
    case 'cancelled':
      return 'Completed';
    default:
      return 'Upcoming';
  }
}

function mapType(t: string): QimochiType {
  switch (t) {
    case 'TV':
    case 'Movie':
    case 'OVA':
    case 'ONA':
    case 'Special':
      return t;
    case 'Music':
      return 'Special';
    case 'Unknown':
    default:
      return 'TV';
  }
}

function titleCase(s: string): string {
  return s
    .split(/[-_\s]+/)
    .filter(Boolean)
    .map((w) => w.charAt(0).toUpperCase() + w.slice(1).toLowerCase())
    .join(' ');
}

function resolveReleaseDate(
  aired: { from?: Date } | undefined,
  year: number | undefined
): Date {
  if (aired?.from instanceof Date) return aired.from;
  if (year) return new Date(Date.UTC(year, 0, 1));
  return new Date(0);
}

function resolveStudio(studios: string[] | undefined): string {
  const first = studios?.[0];
  if (!first) return 'Unknown';
  return titleCase(first);
}

function resolveGenre(genres: string[] | undefined): string[] {
  if (!genres || genres.length === 0) return ['Unknown'];
  return genres.map(titleCase);
}

function resolveRating(stats: { score?: number } | undefined): number {
  const s = stats?.score;
  if (typeof s === 'number' && s >= 0) return s;
  return 0;
}

async function hydrateOne(anime: AnimeEntry): Promise<HydratedAnime> {
  const slug = anime.id;
  const d = anime.data;

  const releaseDate = resolveReleaseDate(d.aired, d.year);
  const addedAt = d.addedAt ?? releaseDate;
  const updatedAt = d.updatedAt;

  return {
    ...anime,
    data: {
      title: d.title,
      titleEnglish: d.titleEnglish,
      titleNative: d.titleNative,

      cover: d.image ?? '',
      status: mapStatus(d.status),
      type: mapType(d.type),
      genre: resolveGenre(d.genres),
      studio: resolveStudio(d.studios),
      releaseDate,
      addedAt,
      updatedAt,
      rating: resolveRating(d.stats),

      year: d.year,
      malId: d.malId,
      banner: d.banner,
      trailer: d.trailer,
      duration: d.duration,

      episodes: episodesBySlug[slug] ?? [],
      franchises: franchisesBySlug[slug] ?? [],
    },
  };
}

export async function getAllAnime(): Promise<HydratedAnime[]> {
  const all = await getCollection('anime', ({ data }) => !data.draft);
  return Promise.all(all.map(hydrateOne));
}

export async function getAnimeById(
  id: string
): Promise<HydratedAnime | null> {
  const anime = await getEntry('anime', id);
  if (!anime || anime.data.draft) return null;
  return hydrateOne(anime);
}

export function sortByRecent(items: HydratedAnime[]): HydratedAnime[] {
  return [...items].sort(
    (a, b) => b.data.addedAt.getTime() - a.data.addedAt.getTime()
  );
}

export function getYear(anime: HydratedAnime): number {
  return anime.data.releaseDate.getFullYear();
}

export interface EpisodePathItem {
  slug: string;
  anime: HydratedAnime;
  episode: EpisodeData;
  episodeIndex: number;
  totalEpisodes: number;
}

export async function getAllEpisodePaths(): Promise<EpisodePathItem[]> {
  const all = await getAllAnime();
  const paths: EpisodePathItem[] = [];

  for (const anime of all) {
    const eps = anime.data.episodes;
    const total = eps.length;

    eps.forEach((episode, episodeIndex) => {
      paths.push({
        slug: anime.id,
        anime,
        episode,
        episodeIndex,
        totalEpisodes: total,
      });
    });
  }

  return paths;
}

export function findEpisode(

/**
 * Ambil artikel terkait berdasarkan tag yang sama.
 * Return max `limit` artikel, exclude artikel itu sendiri.
 */
export function getRelatedPosts(
  current: BlogEntry,
  all: BlogEntry[],
  limit = 3
): BlogEntry[] {
  const currentTags = new Set(current.data.tags);

  if (currentTags.size === 0) return [];

  return all
    .filter(post => post.id !== current.id)
    .map(post => ({
      post,
      score: post.data.tags.filter(t => currentTags.has(t)).length,
    }))
    .filter(({ score }) => score > 0)
    .sort((a, b) => b.score - a.score)
    .slice(0, limit)
    .map(({ post }) => post);
}

/**
 * Ambil semua tag unik dari artikel published + jumlah per tag.
 * Sort by count descending, lalu alfabetis.
 */
export function collectTags(
  posts: BlogEntry[]
): { tag: string; count: number }[] {
  const counts: Record<string, number> = {};

  posts.forEach(post => {
    post.data.tags.forEach(tag => {
      counts[tag] = (counts[tag] || 0) + 1;
    });
  });

  return Object.entries(counts)
    .map(([tag, count]) => ({ tag, count }))
    .sort((a, b) => b.count - a.count || a.tag.localeCompare(b.tag, 'id-ID'));
}

export function filterByTag(posts: BlogEntry[], tag: string): BlogEntry[] {

export function filterByCategory(

  category: string
): BlogEntry[] {
  return posts.filter(post => post.data.category === category);

<script>
  (function () {
    const API = '/api/anime.json';
    const BATCH = 24;
    const DEBOUNCE_MS = 120;
    const MIN_QUERY = 2;
    const FUZZY_THRESHOLD = 0.55;

    type IndexItem = {
      i: string; t: string; c: string;
      s: string; y: string; e: number;
    };

    type Entry = {
      item: IndexItem;
      search: string;
    };

    const form = document.getElementById('search-form') as HTMLFormElement;
    const input = document.getElementById('search-input') as HTMLInputElement;
    const clearBtn = document.getElementById('search-clear') as HTMLButtonElement;
    const emptyState = document.getElementById('search-empty')!;
    const status = document.getElementById('search-status')!;
    const results = document.getElementById('search-results')!;
    const loadMoreWrap = document.getElementById('load-more-wrap')!;
    const loadMoreBtn = document.getElementById('load-more-btn') as HTMLButtonElement;
    const loadMoreText = document.getElementById('load-more-text')!;
    const sentinel = document.getElementById('load-sentinel')!;
    const tmpl = document.getElementById('anime-card-template') as HTMLTemplateElement | null;

    if (!form || !tmpl) return;

    let index: Entry[] = [];
    let matched: IndexItem[] = [];
    let visibleCount = 0;
    let ready = false;

    function levenshtein(a: string, b: string): number {
      if (a === b) return 0;
      if (a.length === 0) return b.length;
      if (b.length === 0) return a.length;

      const m: number[][] = [];
      for (let i = 0; i <= b.length; i++) m[i] = [i];
      for (let j = 0; j <= a.length; j++) m[0][j] = j;

      for (let i = 1; i <= b.length; i++) {
        for (let j = 1; j <= a.length; j++) {
          const cost = a[j - 1] === b[i - 1] ? 0 : 1;
          m[i][j] = Math.min(
            m[i - 1][j] + 1,
            m[i][j - 1] + 1,
            m[i - 1][j - 1] + cost
          );
        }
      }
      return m[b.length][a.length];
    }

    function fuzzyScore(q: string, t: string): number {
      if (!q || !t) return 0;

      const words = t.split(' ');
      let best = 0;
      for (const w of words) {
        const dist = levenshtein(q, w);
        const max = Math.max(q.length, w.length);
        const sim = 1 - dist / max;
        if (sim > best) best = sim;
      }
      return best;
    }
    function statusVariant(s: string): string {
      if (s === 'Ongoing') return 'badge-success';
      if (s === 'Hiatus') return 'badge-warning';
      return 'badge-default';
    }

    function makeCard(item: IndexItem): HTMLElement {
      const node = tmpl!.content.firstElementChild!.cloneNode(true) as HTMLElement;
      node.setAttribute('href', `/anime/${item.i}/`);
      node.dataset.animeTitle = item.t;

      const img = node.querySelector('[data-cover-img]') as HTMLImageElement;
      if (item.c) {
        img.src = item.c;
        img.alt = `Cover ${item.t}`;
      }

      (node.querySelector('.type-badge') as HTMLElement).textContent = item.y;

      const badge = node.querySelector('[data-badge]') as HTMLElement;
      badge.classList.add(statusVariant(item.s));
      badge.textContent = item.s;

      (node.querySelector('h3') as HTMLElement).textContent = item.t;

      const eps = node.querySelector('.episodes') as HTMLElement;
      if (item.e > 0) {
        eps.textContent = `Ep ${item.e}`;
        eps.hidden = false;
      }

      return node;
    }

    function renderNext(): void {
      const end = Math.min(visibleCount + BATCH, matched.length);
      if (end <= visibleCount) return;

      const frag = document.createDocumentFragment();
      for (let i = visibleCount; i < end; i++) {
        frag.appendChild(makeCard(matched[i]!));
      }
      results.appendChild(frag);
      visibleCount = end;
      updateLoadMore();
    }

    function updateLoadMore(): void {
      const remaining = matched.length - visibleCount;
      loadMoreWrap.hidden = remaining <= 0;
      if (remaining > 0) {
        loadMoreText.textContent = `Muat Lebih Banyak (${remaining} lagi)`;
      }
    }

    function resetResults(): void {
      results.innerHTML = '';
      visibleCount = 0;
      matched = [];
    }

    /* ---------- FILTER ---------- */
    function runFilter(qRaw: string): IndexItem[] {
      const q = qRaw.toLowerCase().trim();
      if (!q) return [];

      const substring: IndexItem[] = [];
      for (const entry of index) {
        if (entry.search.includes(q)) substring.push(entry.item);
      }

      if (substring.length > 0) {
        substring.sort((a, b) => {
          const at = a.t.toLowerCase();
          const bt = b.t.toLowerCase();
          const ai = at.indexOf(q);
          const bi = bt.indexOf(q);
          if (ai !== bi) return ai - bi;
          return at.length - bt.length;
        });
        return substring;
      }

      const fuzzy = index
        .map(e => ({ item: e.item, score: fuzzyScore(q, e.search) }))
        .filter(({ score }) => score >= FUZZY_THRESHOLD)
        .sort((a, b) => b.score - a.score)
        .map(s => s.item);

      return fuzzy;
    }

    function applyFilter(query: string): void {
      const q = query.trim();

      if (!ready) return;

      if (q.length === 0) {
        resetResults();
        emptyState.hidden = false;
        status.hidden = true;
        results.hidden = true;
        loadMoreWrap.hidden = true;
        return;
      }

      if (q.length < MIN_QUERY) {
        resetResults();
        emptyState.hidden = true;
        status.textContent = `Ketik minimal ${MIN_QUERY} karakter.`;
        status.hidden = false;
        results.hidden = true;
        loadMoreWrap.hidden = true;
        return;
      }

      resetResults();
      matched = runFilter(q);

      emptyState.hidden = true;
      results.hidden = false;

      if (matched.length === 0) {
        status.textContent = `Tidak ada hasil untuk "${q}".`;
        status.hidden = false;
        results.hidden = true;
        loadMoreWrap.hidden = true;
        return;
      }

      status.textContent = `${matched.length} hasil untuk "${q}"`;
      status.hidden = false;
      renderNext();
    }

    /* ---------- EVENTS ---------- */
    let debounceId: number | null = null;
    function debouncedFilter(q: string): void {
      if (debounceId !== null) window.clearTimeout(debounceId);
      debounceId = window.setTimeout(() => applyFilter(q), DEBOUNCE_MS);
    }

    function updateClearBtn(): void {
      clearBtn.hidden = input.value.length === 0;
    }

    input.addEventListener('input', () => {
      updateClearBtn();
      debouncedFilter(input.value);
    });
    updateClearBtn();

    clearBtn.addEventListener('click', () => {
      input.value = '';
      input.focus();
      updateClearBtn();
      applyFilter('');
      const url = new URL(window.location.href);
      url.searchParams.delete('q');
      window.history.pushState({}, '', url);
    });

    form.addEventListener('submit', e => {
      e.preventDefault();
      if (debounceId !== null) window.clearTimeout(debounceId);
      const q = input.value.trim();
      const url = new URL(window.location.href);
      if (q) url.searchParams.set('q', q);
      else url.searchParams.delete('q');
      window.history.pushState({}, '', url);
      applyFilter(q);
    });

    loadMoreBtn.addEventListener('click', () => renderNext());

    if ('IntersectionObserver' in window) {
      const io = new IntersectionObserver(entries => {
        entries.forEach(entry => {
          if (!entry.isIntersecting) return;
          if (matched.length - visibleCount <= 0) return;
          renderNext();
        });
      }, { rootMargin: '400px 0px' });
      io.observe(sentinel);
    }

    window.addEventListener('popstate', () => {
      const p = new URLSearchParams(window.location.search);
      const q = p.get('q') || '';
      input.value = q;
      updateClearBtn();
      if (debounceId !== null) window.clearTimeout(debounceId);
      applyFilter(q);
    });

    /* ---------- INIT (fetch JSON) ---------- */
    const params = new URLSearchParams(window.location.search);
    const initialQuery = params.get('q') || '';
    input.value = initialQuery;
    updateClearBtn();

    fetch(API, { headers: { Accept: 'application/json' } })
      .then(res => {
        if (!res.ok) throw new Error(`HTTP ${res.status}`);
        return res.json() as Promise<IndexItem[]>;
      })
      .then(json => {
        if (!Array.isArray(json)) {
          console.error('[Search] Invalid payload');
          return;
        }

        index = json.map(item => ({
          item,
          search: (item.t || '').toLowerCase(),
        }));

        ready = true;

        if (initialQuery.trim().length >= MIN_QUERY) {
          applyFilter(initialQuery);
        } else {
          applyFilter('');
        }
      })
      .catch(err => {
        console.error('[Search] fetch failed:', err);
        status.textContent = 'Gagal memuat data. Coba refresh halaman.';
        status.hidden = false;
      });
  })();
</script>
